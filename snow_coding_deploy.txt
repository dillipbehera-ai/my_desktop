from airflow import DAG
from airflow.operators.bash import BashOperator
from airflow.providers.amazon.aws.transfers.local_to_s3 import LocalFilesystemToS3Operator
from airflow.providers.snowflake.operators.snowflake import SnowflakeOperator
from datetime import datetime

default_args = {"start_date": datetime(2024, 1, 1)}

with DAG("unix_s3_snowflake_pipeline", schedule_interval="@daily", catchup=False, default_args=default_args) as dag:

    upload_to_s3 = LocalFilesystemToS3Operator(
        task_id='upload_csv_to_s3',
        filename='/data/claims_2025.csv',
        dest_key='claims/claims_2025.csv',
        dest_bucket='raw-zone-claims',
        aws_conn_id='aws_default',
        replace=True
    )

    load_to_snowflake = SnowflakeOperator(
        task_id='copy_into_snowflake',
        sql="""
        COPY INTO raw_db.claims_stg FROM @s3_stage/claims FILE_FORMAT = (FORMAT_NAME = 'CSV_FORMAT');
        """,
        snowflake_conn_id='snowflake_default'
    )

    upload_to_s3 >> load_to_snowflake

#!/bin/bash

# Variables (use --option or config profiles for better security)
USER="your_user"
ACCOUNT="your_account"
WAREHOUSE="your_warehouse"
DATABASE="your_database"
SCHEMA="your_schema"
SQL_FILE="snow_job.sql"
LOG_FILE="snow_run.log"

echo "▶️ Running Snowflake SQL file: $SQL_FILE"

snowsql -a $ACCOUNT -u $USER -w $WAREHOUSE -d $DATABASE -s $SCHEMA -f $SQL_FILE -o log_level=INFO -o output_file=$LOG_FILE

EXIT_CODE=$?

if [ $EXIT_CODE -eq 0 ]; then
    echo "✅ SQL script executed successfully."
else
    echo "❌ SQL script failed. Check log: $LOG_FILE"
    exit $EXIT_CODE
fi

-- snow_job.sql

-- Delete existing test data
DELETE FROM my_schema.customer_dim WHERE updated_at < CURRENT_DATE - 30;

-- Insert new data
INSERT INTO my_schema.customer_dim (cust_id, name, email, updated_at)
VALUES (1, 'Alice', 'alice@email.com', CURRENT_TIMESTAMP);

-- Merge UPSERT logic
MERGE INTO my_schema.customer_dim AS target
USING (SELECT 1 AS cust_id, 'Alice Smith' AS name, 'alice@email.com' AS email, CURRENT_TIMESTAMP AS updated_at) AS src
ON target.cust_id = src.cust_id
WHEN MATCHED THEN UPDATE SET target.name = src.name, target.updated_at = src.updated_at
WHEN NOT MATCHED THEN INSERT (cust_id, name, email, updated_at)
VALUES (src.cust_id, src.name, src.email, src.updated_at);

-- Create or Replace Procedure with Error Handling
CREATE OR REPLACE PROCEDURE my_schema.proc_sample_update()
RETURNS STRING
LANGUAGE SQL
AS
$$
BEGIN
    -- Update statement
    UPDATE my_schema.customer_dim
    SET email = LOWER(email)
    WHERE email LIKE '%EMAIL.COM';

    RETURN 'Procedure executed successfully';
EXCEPTION
    WHEN STATEMENT_ERROR THEN
        RETURN 'Failed in statement: ' || ERROR_MESSAGE();
    WHEN OTHER THEN
        RETURN 'General error: ' || ERROR_MESSAGE();
END;
$$;

-- Call the Procedure
CALL my_schema.proc_sample_update();

import snowflake.connector

# ❄️ Snowflake connection parameters
conn = snowflake.connector.connect(
    user='YOUR_USERNAME',
    password='YOUR_PASSWORD',
    account='YOUR_ACCOUNT_ID',
    warehouse='YOUR_WAREHOUSE',
    database='YOUR_DATABASE',
    schema='my_schema'
)

cursor = conn.cursor()

try:
    with open("snow_job.sql", "r") as f:
        sql_script = f.read()

    # Split by semicolon to execute multiple statements
    for statement in sql_script.strip().split(";"):
        if statement.strip():
            print(f"Executing: {statement[:50]}...")
            cursor.execute(statement)

except Exception as e:
    print(f"❌ Error running SQL script: {e}")

finally:
    cursor.close()
    conn.close()

✅ 3. Stored Procedure: SCD Type 2 Merge Logic✅✅✅✅
sql
Copy
Edit
CREATE OR REPLACE PROCEDURE SP_LOAD_CORE_CUSTOMER()
RETURNS STRING
LANGUAGE SQL
AS
$$
DECLARE
  rows_inserted INT;
  rows_updated INT;
BEGIN
  -- Step 1: Expire old records
  UPDATE CORE_CUSTOMER tgt
  SET valid_to = CURRENT_DATE - 1,
      is_active = 'N'
  FROM STG_CUSTOMER src
  WHERE tgt.cust_id = src.cust_id
    AND tgt.is_active = 'Y'
    AND (
      tgt.name != src.name OR tgt.email != src.email
    );

  LET rows_updated = SQLROWCOUNT;

  -- Step 2: Insert new versions
  INSERT INTO CORE_CUSTOMER (cust_id, name, email, valid_from, valid_to, is_active)
  SELECT 
    cust_id,
    name,
    email,
    CURRENT_DATE,  -- valid_from
    NULL,          -- valid_to (open-ended)
    'Y'
  FROM STG_CUSTOMER src
  LEFT JOIN CORE_CUSTOMER tgt
    ON src.cust_id = tgt.cust_id AND tgt.is_active = 'Y'
  WHERE tgt.cust_id IS NULL OR tgt.name != src.name OR tgt.email != src.email;

  LET rows_inserted = SQLROWCOUNT;

  RETURN 'Inserted: ' || rows_inserted || ', Updated: ' || rows_updated;
END;
$$;

-- Step 1: Expire target rows if changes found in source
UPDATE TRG_CUSTOMER_HIST tgt
SET valid_to = CURRENT_DATE - 1,
    is_active = 'N'
FROM SRC_CUSTOMER_HIST src
WHERE tgt.cust_id = src.cust_id
  AND tgt.is_active = 'Y'
  AND src.is_active = 'Y'
  AND (
    tgt.name != src.name OR tgt.email != src.email
  );

-- Step 2: Insert changed records into target
INSERT INTO TRG_CUSTOMER_HIST (cust_id, name, email, valid_from, valid_to, is_active)
SELECT 
  src.cust_id,
  src.name,
  src.email,
  CURRENT_DATE,      -- valid_from
  NULL,              -- valid_to
  'Y'                -- is_active
FROM SRC_CUSTOMER_HIST src
LEFT JOIN TRG_CUSTOMER_HIST tgt
  ON src.cust_id = tgt.cust_id AND tgt.is_active = 'Y'
WHERE src.is_active = 'Y'
  AND (tgt.cust_id IS NULL OR tgt.name != src.name OR tgt.email != src.email);
✅ Summary of Logic:✅✅✅✅✅✅✅✅✅✅✅✅✅✅✅✅
Step	Description
🔁 Compare active versions	Only is_active = 'Y' rows are compared
❌ Expire changed rows	Sets valid_to in target if difference found
➕ Insert new rows	Adds latest version as is_active = 'Y'
🔄 Historical preserved	Older versions remain with is_active = 'N'
